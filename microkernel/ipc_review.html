硬化设计要点

机制防什么实现位置
reply_id per slot伪造 reply / 错配 inflight send / 静默丢 replysend 入队时内核分配 token → reply 时必须匹配才能解
send_wait / recv_wait 精确隔离旧版一个 blocked bool 互相污染 → reply 误解 recv 阻塞 / receive 误解 send 阻塞每个 TCB 独立两个 bool + 各自唯一的唤醒函数
队列永不溢出send 同步语义 + QUEUE_SIZE = MAX_TASKS → 每 mailbox 积压 ≤ 31 条assert 防御，无 while 阻塞（背压是死代码）
无 silent drop旧版"队列满丢最老"违反硬化目标reply 积压最多 1 条，q_push 永不溢出
服务者死亡清理服务死了但 client send_wait 永远阻塞 → 系统卡死扫 inflight + 推 synthetic error reply + wake_send_waiter

  1. 1/*
  2. 2 * ipc.c — IPC 硬化版
  3. 3 *
  4. 4 * 硬化相对于原型的 5 个改动（+1 个设计简化）：
  5. 5 *
  6. 6 * P0 | reply_id per slot：每条 send 分配唯一 token；reply 必须带匹配 token 才能解
  7. 7 * P0 | 两种独立阻塞原因：send_wait / recv_wait，精确隔离（旧版一个 bool 互相污染）
  8. 8 * P1 | reply 校验 reply_id 合法性：reply_id 没匹配到任何 inflight → MK_ERR_INVALID
  9. 9 * P1 | 单一真相源：删 ipc_has_msg / ipc_blocked / ipc_waiter
  10. 10 * P2 | from / reply_id 强制内核覆盖，用户写的作废
  11. 11 *
  12. 12 * 设计简化：为什么删 space_wait（队列满阻塞）？
  13. 13 * send 是同步的 → 每个任务同时最多 1 条 inflight send。
  14. 14 * QUEUE_SIZE = MAX_TASKS → 一个 mailbox 最多积压 N-1 条 send（其他各 1 条）。
  15. 15 * N 槽 ≥ N-1 积压 → 永远差 1 格，队列不可能满。
  16. 16 * 背压是死代码。用 assert 防御而非 while 阻塞。
  17. 17 *
  18. 18 * 精确隔离的阻塞原因为什么重要：
  19. 19 * 旧版只有一个 ipc_blocked bool + state=BLOCKED。wake_unblocked 不管什么原因都解，
  20. 20 * 导致 reply 可能误解一个 recv 阻塞的任务（白调度）；receive 广播唤醒又可能解 send
  21. 21 * 阻塞的任务（状态污染）。拆成两个 bool + 精确匹配唤醒后，每种阻塞只有对应的
  22. 22 * 那一种原语能解，零误触。
  23. 23 */

  41. 41/* ================================================================
  42. 42 * 环形队列 —— 返回值硬化
  43. 43 * q_push: 返回写入的 slot index（后续要登记 inflight）
  44. 44 * q_pop: 返回弹出的 slot index（当前不返回给外部，保留 API 对称）
  45. 45 *
  46. 46 * 不变量（由同步 send 语义保证）：

  48. 48 * QUEUE_SIZE = N → q_push 永远不会溢出。用 assert 防御，不用 while 阻塞。
  49. 49 * ================================================================ */
  50. 50static inline bool q_empty(const mk_mailbox_t *m) { return m->count == 0; }
  51. 51
  52. 52/* L1 优化：static inline + 位与代替模（QUEUE_SIZE=32=2^5 → mask=31）
  53. 53 * 编译器 -O2 可能已经自动做了，但显式写出来防止退化 */
  54. 54static inline uint8_t q_push(mk_mailbox_t *m, const mk_msg_t *msg)
  55. 55{
  56. 56 uint8_t slot = m->tail;
  57. 57 m->slots[slot] = *msg;
  58. 58 m->tail = (uint8_t)((slot + 1) & (MK_IPC_QUEUE_SIZE - 1));
  59. 59 m->count++;
  60. 60 return slot;
  61. 61}

  63. 63static inline uint8_t q_pop(mk_mailbox_t *m, mk_msg_t *out)
  64. 64{
  65. 65 uint8_t slot = m->head;
  66. 66 if (out) *out = m->slots[slot];
  67. 67 m->head = (uint8_t)((slot + 1) & (MK_IPC_QUEUE_SIZE - 1));
  68. 68 m->count--;
  69. 69 return slot;
  70. 70}

  98. 98/* ================================================================
  99. 99 * send — 硬化版
  100. 100 *
  101. 101 * 分配 reply token → 内核写 from/reply_id → 登记 inflight → 入队
  102. 102 * 自己 send_wait 阻塞等 reply
  103. 103 *
  104. 104 * 不变量：同步 send + QUEUE_SIZE = N → q_push 永远不溢出。
  105. 105 * 无 silent drop，reply_id 保证 reply 能对应回这条 send
  106. 106 * ================================================================ */
  107. 107mk_err_t mk_ipc_send(uint8_t to, const mk_msg_t *msg)
  108. 108{
  109. 109 uint8_t self = mk_current_tid();
  110. 110 if (to >= MK_MAX_TASKS || to == self) return MK_ERR_INVALID;
  111. 111 if (!msg) return MK_ERR_INVALID;
  112. 112
  113. 113 mk_tcb_t *dst_tcb = mk_tcb_get(to);
  114. 114 mk_mailbox_t *dst_mbox = &g_mboxes[to];
  115. 115 mk_tcb_t *me = mk_tcb_get(self);
  116. 116
  117. 117 if (!dst_tcb || dst_tcb->state == MK_TASK_DEAD) return MK_ERR_NOIPC;
  118. 118
  119. 119 uint16_t reply_id = ++g_reply_counter;
  120. 120 mk_msg_t full_msg = *msg;
  121. 121 full_msg.from = self;
  122. 122 full_msg.reply_id = reply_id;
  123. 123
  124. 124 uint8_t slot = q_push(dst_mbox, &full_msg);
  125. 125 dst_mbox->inflight[slot] = true;
  126. 126 dst_mbox->reply_id_of_slot[slot] = reply_id;
  127. 127
  128. 128 wake_recv_waiter(to);
  129. 129
  130. 130 me->ipc_send_wait = true;
  131. 131 mk_tcb_set_state(me, MK_TASK_BLOCKED);
  132. 132 mk_sched_unready(self);
  133. 133
  134. 134 mk_sched_tick();
  135. 135
  136. 136 me->ipc_send_wait = false;
  137. 137 return MK_OK;
  138. 138}

  140. 140/* ================================================================
  141. 141 * receive — 硬化版
  142. 142 *
  143. 143 * 队空 → recv_wait 阻塞；队列有消息 → pop
  144. 144 * 被唤醒后循环检查队列（防御多次 wake 间队列被消耗空）
  145. 145 *
  146. 146 * 注意：**receive 不清 inflight**！
  147. 147 * inflight 表示 "这条 send 还在等 reply"，不是 "还在队列里"。
  148. 148 * send 消息被 receive 取走只是让服务者看到了请求，
  149. 149 * 但 send 者的 send_wait 还在阻塞等 reply。
  150. 150 * inflight 只能由 reply() 匹配成功后清。
  151. 151 * ================================================================ */
  152. 152mk_err_t mk_ipc_receive(mk_msg_t *msg)
  153. 153{
  154. 154 if (!msg) return MK_ERR_INVALID;
  155. 155
  156. 156 uint8_t tid = mk_current_tid();
  157. 157 mk_tcb_t *tcb = mk_tcb_get(tid);
  158. 158 mk_mailbox_t *mbox = &g_mboxes[tid];
  159. 159
  160. 160 while (q_empty(mbox)) {
  161. 161 tcb->ipc_recv_wait = true;
  162. 162 mk_tcb_set_state(tcb, MK_TASK_BLOCKED);
  163. 163 mk_sched_unready(tid);
  164. 164
  165. 165 mk_sched_tick();
  166. 166 }
  167. 167
  168. 168 q_pop(mbox, msg);
  169. 169 return MK_OK;
  170. 170}

  172. 172/* ================================================================
  173. 173 * reply — 硬化版
  174. 174 *
  175. 175 * 1. 在自己的 mailbox 里找 reply_id 匹配的 inflight slot → 清掉
  176. 176 * 这是 "这笔账我结了" 的意思。没匹配到 → 返回 MK_ERR_INVALID。
  177. 177 * 2. 投递 reply 消息到对方 inbox（解 recv_wait）。
  178. 178 * 不变量：对方同时最多 1 条 inflight send，reply 积压最多 1 条。
  179. 179 * 3. 解对方的 send_wait —— 那个被我们匹配到 inflight 的 send 者。
  180. 180 *
  181. 181 * reply 和 send 的根本区别：reply 不阻塞，立即返回。
  182. 182 * send 则是投递消息 + send_wait 阻塞等 reply 解。
  183. 183 *
  184. 184 * 硬化：reply 不允许 silent drop。旧版 "队列满丢最老" 违反硬化目标，
  185. 185 * 也永远不会触发（同步 send 下 reply 积压最多 1 条）。
  186. 186 * ================================================================ */
  187. 187mk_err_t mk_ipc_reply(uint8_t to, const mk_msg_t *msg)
  188. 188{
  189. 189 uint8_t self = mk_current_tid();
  190. 190 if (to >= MK_MAX_TASKS || to == self) return MK_ERR_INVALID;
  191. 191 if (!msg) return MK_ERR_INVALID;
  192. 192
  193. 193 /* P0-2 硬化：reply_id==0 直接拒，不扫 32 个 slot。
  194. 194 * g_reply_counter 从 1 开始，0 永远不可能是合法 token。
  195. 195 * 开发者忘了写 reply.reply_id = req.reply_id 时会撞到这里。 */
  196. 196 if (msg->reply_id == 0) return MK_ERR_INVALID;
  197. 197
  198. 198 mk_tcb_t *to_tcb = mk_tcb_get(to);
  199. 199 mk_mailbox_t *dst_mbox = &g_mboxes[to]; /* reply 目标的 mailbox (send 者的 inbox) */
  200. 200 mk_mailbox_t *my_mbox = &g_mboxes[self]; /* 我 reply 者自己的 mailbox */
  201. 201
  202. 202 if (!to_tcb || to_tcb->state == MK_TASK_DEAD) return MK_ERR_NOIPC;
  203. 203
  204. 204 /* ---- 关键校验：reply_id 必须匹配我 mailbox 里某个 inflight slot ---- */
  205. 205 bool inflight_found = false;
  206. 206
  207. 207 for (uint8_t i = 0; i < MK_IPC_QUEUE_SIZE; ++i) {
  208. 208 if (my_mbox->inflight[i]
  209. 209 && my_mbox->reply_id_of_slot[i] == msg->reply_id) {
  210. 210 my_mbox->inflight[i] = false;
  211. 211 my_mbox->reply_id_of_slot[i] = 0;
  212. 212 inflight_found = true;
  213. 213 break;
  214. 214 }
  215. 215 }
  216. 216
  217. 217 if (!inflight_found) {
  218. 218 return MK_ERR_INVALID;
  219. 219 }
  220. 220
  221. 221 /* ---- 投递 reply 消息到对方 inbox ---- */
  222. 222 mk_msg_t full_msg = *msg;
  223. 223 full_msg.from = self;
  224. 224
  225. 225 q_push(dst_mbox, &full_msg);
  226. 226
  227. 227 wake_send_waiter(to);
  228. 228
  229. 229 return MK_OK;
  230. 230}

  238. 238/* ================================================================
  239. 239 * mk_ipc_cleanup_dead_service — 服务者死亡时清理它的 IPC 状态
  240. 240 *
  241. 241 * 必须在 mk_task_exit 调 —— 否则：
  242. 242 * client 正在 send_wait 等这个服务者 reply
  243. 243 * 服务者死了 → inflight 永远不被清 → 永远不 wake_send_waiter
  244. 244 * → 那些 client 永久 BLOCKED，整个系统卡死
  245. 245 *
  246. 246 * 两步：
  247. 247 * 1. 扫 inflight[] 里所有 slot —— 每个对应一个还在等 reply 的 client
  248. 248 * 给每个 client 推一条 synthetic error reply（tag=0 表示"服务已死"）
  249. 249 * 调 wake_send_waiter 解它的阻塞
  250. 250 * 2. 清空 mailbox 队列 —— 时刻 A（send 刚入队还没被 receive）的残留
  251. 251 * 这些消息的 inflight 已经被上面处理过了，队列只是留着脏数据
  252. 252 * 下次 tid 被重用时脏队列会引发错觉
  253. 253 * ================================================================ */
  254. 254void mk_ipc_cleanup_dead_service(uint8_t dead_tid)
  255. 255{
  256. 256 if (dead_tid >= MK_MAX_TASKS) return;
  257. 257
  258. 258 mk_mailbox_t *my = &g_mboxes[dead_tid];
  259. 259
  260. 260 for (uint8_t i = 0; i < MK_IPC_QUEUE_SIZE; ++i) {
  261. 261 if (!my->inflight[i]) continue;
  262. 262
  263. 263 uint8_t client = my->slots[i].from;
  264. 264 uint16_t reply_id = my->reply_id_of_slot[i];
  265. 265
  266. 266 /* 清 inflight（我死了，这笔账结不了） */
  267. 267 my->inflight[i] = false;
  268. 268 my->reply_id_of_slot[i] = 0;
  269. 269
  270. 270 /* 防御性边界：from 必须是合法 tid */
  271. 271 if (client >= MK_MAX_TASKS) continue;
  272. 272
  273. 273 /* 给 client 推 synthetic reply —— tag=0 约定"服务已死"
  274. 274 * reply_id 必须是那个 client send 时内核分配的 token，
  275. 275 * 这样 client.unblock() 后 receive() 能正常读到 */
  276. 276 mk_msg_t err;
  277. 277 memset(&err, 0, sizeof(err));
  278. 278 err.reply_id = reply_id;
  279. 279 err.from = dead_tid; /* 内核填，虽然 reply_id 更关键 */
  280. 280 /* err.data[0] = MK_ERR_NOIPC = -1 —— client 的 mk_ipc_send 返回后
  281. 281 * 应该 receive 这条消息时能看到服务已死 */
  282. 282 err.data[0] = MK_ERR_NOIPC;
  283. 283
  284. 284 mk_mailbox_t *dst = &g_mboxes[client];
  285. 285 q_push(dst, &err);
  286. 286
  287. 287 /* 解 client 的 send_wait —— 相当于 reply() 做的事 */
  288. 288 wake_send_waiter(client);
  289. 289 }
  290. 290
  291. 291 /* 清空残留队列 —— 时刻 A 的 send 还没被 receive */
  292. 292 while (!q_empty(my)) {
  293. 293 q_pop(my, NULL);
  294. 294 }
  295. 295}
